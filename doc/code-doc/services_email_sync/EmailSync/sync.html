Future<void> sync()

Recharge le compte connecté et met à jour users/{uid}.email si l'adresse a changé. Ne lève jamais d'exception.

Implementation

static Future<void> sync() async {
  if (_running) return;
  _running = true;
  try {
    final User? user = FirebaseAuth.instance.currentUser;
    if (user == null) return;
    await user.reload();
    final User? refreshed = FirebaseAuth.instance.currentUser;
    if (refreshed == null) return;
    final DocumentReference<Map<String, dynamic>> ref = FirebaseFirestore
        .instance
        .collection('users')
        .doc(refreshed.uid);
    final DocumentSnapshot<Map<String, dynamic>> snapshot = await ref.get();
    if (!snapshot.exists) return;
    final Object? stored = snapshot.data()?['email'];
    final String? email = emailToWrite(
      authEmail: refreshed.email,
      storedEmail: stored is String ? stored : null,
    );
    if (email != null) await ref.update(<String, dynamic>{'email': email});
  } catch (e, st) {
    developer.log(
      'Synchronisation de l\'adresse e-mail impossible',
      name: 'EmailSync',
      error: e,
      stackTrace: st,
    );
  } finally {
    _running = false;
  }
}
