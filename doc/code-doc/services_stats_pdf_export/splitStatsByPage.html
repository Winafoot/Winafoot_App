Découpe stats (dans l'ordre du catalogue) en pages de maxPerPage colonnes au plus. Les catégories consécutives se regroupent sur une même page ; une catégorie qui ne rentre pas dans la place restante ouvre la page suivante, sauf si elle dépasse à elle seule maxPerPage : elle est alors coupée (et remplit la page en cours).

Implementation

@visibleForTesting
List<List<StatDef>> splitStatsByPage(
  List<StatDef> stats, {
  int maxPerPage = kStatsPerPage,
}) {
  final List<List<StatDef>> pages = <List<StatDef>>[];
  List<StatDef> page = <StatDef>[];
  int start = 0;
  while (start < stats.length) {
    int end = start;
    while (end < stats.length && stats[end].category == stats[start].category) {
      end++;
    }
    final List<StatDef> run = stats.sublist(start, end);
    start = end;
    if (run.length <= maxPerPage && page.length + run.length > maxPerPage) {
      pages.add(page);
      page = <StatDef>[];
    }
    for (final StatDef s in run) {
      if (page.length == maxPerPage) {
        pages.add(page);
        page = <StatDef>[];
      }
      page.add(s);
    }
  }
  if (page.isNotEmpty) pages.add(page);
  return pages;
}
