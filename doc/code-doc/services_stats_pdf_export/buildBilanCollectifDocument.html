Construit le PDF du « Bilan collectif » (sans le partager) : A4 portrait, la seule period affichée à l'écran (1ère mi-temps, 2ème mi-temps ou Total match), au format du document du client. Si elle ne tient pas sur une page, elle déborde sur la suivante, en-tête répété. theme fournit la police (Helvetica à défaut).

Implementation

pw.Document buildBilanCollectifDocument({
  required GameMatch match,
  required MatchPeriod period,
  required String ownTeam,
  required String opponent,
  pw.ThemeData? theme,
}) {
  final pw.Document doc = pw.Document(theme: theme);
  final List<CollectifBlock> blocks = collectifBlocks(match, period);
  doc.addPage(
    pw.MultiPage(
      pageFormat: PdfPageFormat.a4,
      margin: const pw.EdgeInsets.all(28),
      header: (pw.Context context) => pw.Column(
        crossAxisAlignment: pw.CrossAxisAlignment.stretch,
        children: <pw.Widget>[
          pw.Text(
            'STATISTIQUES COLLECTIVES',
            textAlign: pw.TextAlign.center,
            style: pw.TextStyle(fontSize: 16, fontWeight: pw.FontWeight.bold),
          ),
          pw.SizedBox(height: 8),
          pw.Row(
            crossAxisAlignment: pw.CrossAxisAlignment.start,
            children: <pw.Widget>[
              pw.Expanded(
                child: pw.Text(
                  '$ownTeam / $opponent',
                  style: pw.TextStyle(
                    fontSize: 12,
                    fontWeight: pw.FontWeight.bold,
                  ),
                ),
              ),
              pw.SizedBox(width: 12),
              pw.Text(
                _kPeriodLabels[period]!,
                style: pw.TextStyle(
                  fontSize: 12,
                  fontWeight: pw.FontWeight.bold,
                ),
              ),
            ],
          ),
          pw.SizedBox(height: 10),
        ],
      ),
      build: (pw.Context context) => <pw.Widget>[
        // Une page sans contenu n'est pas écrite : sans ce texte la période
        // disparaîtrait du PDF.
        if (blocks.isEmpty)
          pw.Text(
            'Aucune statistique enregistrée pour ce match.',
            style: const pw.TextStyle(fontSize: 11),
          ),
        for (int i = 0; i < blocks.length; i += 2)
          pw.Padding(
            padding: const pw.EdgeInsets.only(bottom: 10),
            child: pw.Row(
              crossAxisAlignment: pw.CrossAxisAlignment.start,
              children: <pw.Widget>[
                pw.Expanded(child: _collectifBlock(blocks[i])),
                pw.SizedBox(width: 24),
                pw.Expanded(
                  child: i + 1 < blocks.length
                      ? _collectifBlock(blocks[i + 1])
                      : pw.SizedBox(),
                ),
              ],
            ),
          ),
      ],
    ),
  );
  return doc;
}
